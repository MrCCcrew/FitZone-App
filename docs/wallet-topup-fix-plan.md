### src/lib/payments/wallet-settlement.ts (NEW FILE)

```typescript
import { PrismaClient } from "@prisma/client";
import type { AuditLogSource } from "./types";

const db = new PrismaClient();

interface SettleWalletTopupResult {
  success: boolean;
  walletTransactionId?: string;
  error?: string;
  alreadySettled?: boolean;
}

/**
 * Settle wallet_topup payment transaction
 * 
 * IDEMPOTENT: Safe to call multiple times
 * ATOMIC: Uses db.$transaction
 * VALIDATED: Checks all preconditions
 * AUDITED: Logs all actions
 * 
 * @param paymentTransactionId - PaymentTransaction.id
 * @param source - Who triggered this settlement
 * @returns Result object
 */
export async function settleWalletTopup(
  paymentTransactionId: string,
  source: AuditLogSource
): Promise<SettleWalletTopupResult> {
  return await db.$transaction(async (tx) => {
    // Step 1: Read PaymentTransaction
    const payment = await tx.paymentTransaction.findUnique({
      where: { id: paymentTransactionId },
      select: {
        id: true,
        userId: true,
        amount: true,
        currency: true,
        status: true,
        purpose: true,
        settlementStatus: true,
        referenceCode: true,
        providerReference: true,
        externalReference: true,
        walletTransaction: {
          select: { id: true },
        },
      },
    });

    if (!payment) {
      const error = "PaymentTransaction not found";
      await logAudit(tx, paymentTransactionId, "settlement_failed", source, { error });
      return { success: false, error };
    }

    // Step 2: Validate preconditions
    if (payment.status !== "paid") {
      const error = `Cannot settle: status is ${payment.status}, not paid`;
      await logAudit(tx, paymentTransactionId, "settlement_failed", source, { 
        error,
        settlementStatus: payment.settlementStatus 
      });
      return { success: false, error };
    }

    if (payment.purpose !== "wallet_topup") {
      const error = `Cannot settle: purpose is ${payment.purpose}, not wallet_topup`;
      await logAudit(tx, paymentTransactionId, "settlement_failed", source, { error });
      return { success: false, error };
    }

    if (payment.currency !== "EGP") {
      const error = `Cannot settle: currency is ${payment.currency}, not EGP`;
      await logAudit(tx, paymentTransactionId, "settlement_failed", source, { error });
      return { success: false, error };
    }

    if (!payment.userId) {
      const error = "Cannot settle: userId is null";
      await logAudit(tx, paymentTransactionId, "settlement_failed", source, { error });
      return { success: false, error };
    }

    if (payment.amount <= 0) {
      const error = `Cannot settle: amount is ${payment.amount}, must be > 0`;
      await logAudit(tx, paymentTransactionId, "settlement_failed", source, { error });
      return { success: false, error };
    }

    // Step 3: Check if already settled (idempotency)
    if (payment.walletTransaction) {
      await logAudit(tx, paymentTransactionId, "settlement_completed", source, {
        walletTransactionId: payment.walletTransaction.id,
        note: "Already settled (idempotent return)",
      });
      return { 
        success: true, 
        walletTransactionId: payment.walletTransaction.id,
        alreadySettled: true 
      };
    }

    // Step 4: Update settlement status to processing
    await tx.paymentTransaction.update({
      where: { id: paymentTransactionId },
      data: { settlementStatus: "processing" },
    });

    await logAudit(tx, paymentTransactionId, "settlement_started", source, {
      userId: payment.userId,
      amount: payment.amount,
      currency: payment.currency,
    });

    // Step 5: Upsert wallet and increment balance
    const wallet = await tx.wallet.upsert({
      where: { userId: payment.userId },
      update: { balance: { increment: payment.amount } },
      create: { userId: payment.userId, balance: payment.amount },
    });

    // Step 6: Create WalletTransaction with payment link
    const walletTransaction = await tx.walletTransaction.create({
      data: {
        walletId: wallet.id,
        amount: payment.amount,
        type: "credit",
        description: `شحن المحفظة عبر Paymob - ${payment.referenceCode ?? payment.id}`,
        paymentTransactionId: payment.id,
      },
    });

    // Step 7: Mark as settled
    await tx.paymentTransaction.update({
      where: { id: paymentTransactionId },
      data: { 
        settlementStatus: "settled",
        settledAt: new Date(),
        settlementError: null, // Clear any previous error
      },
    });

    await logAudit(tx, paymentTransactionId, "settlement_completed", source, {
      walletTransactionId: walletTransaction.id,
      userId: payment.userId,
      amount: payment.amount,
      currency: payment.currency,
      providerReference: payment.providerReference,
      externalReference: payment.externalReference,
      referenceCode: payment.referenceCode,
    });

    return { 
      success: true, 
      walletTransactionId: walletTransaction.id,
      alreadySettled: false 
    };
  });
}

async function logAudit(
  tx: any,
  paymentTransactionId: string,
  action: AuditLogAction,
  source: AuditLogSource,
  data: {
    error?: string;
    settlementStatus?: string;
    walletTransactionId?: string;
    userId?: string;
    amount?: number;
    currency?: string;
    providerReference?: string | null;
    externalReference?: string | null;
    referenceCode?: string | null;
    note?: string;
  }
) {
  await tx.paymentAuditLog.create({
    data: {
      paymentTransactionId,
      action,
      source,
      settlementStatus: data.settlementStatus ?? null,
      walletTransactionId: data.walletTransactionId ?? null,
      userId: data.userId ?? null,
      amount: data.amount ?? null,
      currency: data.currency ?? null,
      providerReference: data.providerReference ?? null,
      externalReference: data.externalReference ?? null,
      referenceCode: data.referenceCode ?? null,
      errorMessage: data.error || data.note || null,
    },
  });
}
```

### If Reconciliation Fails

- ❌ DO NOT manually UPDATE wallet balance
- ❌ DO NOT manually INSERT WalletTransaction
- ✅ Check logs for error message
- ✅ Fix issue in settleWalletTopup function
- ✅ Re-run reconciliation script (it's idempotent)

1. Stop accepting new wallet_topup payments temporarily
2. Fix bug in `settleWalletTopup`
3. Deploy fix
4. Run reconciliation for all `requires_reconciliation` transactions
5. Re-enable wallet_topup

## 9. Monitoring & Alerts

### Metrics to Track

1. **Settlement Success Rate**
   - wallet_topup paid → settled within 5 minutes: should be > 99%

2. **Stuck Transactions**
   - Alert if `settlementStatus = processing` for > 10 minutes
   - Alert if `settlementStatus = failed` or `requires_reconciliation`

3. **Idempotency Hits**
   - Track how often `alreadySettled = true`
   - High rate = duplicate webhooks issue

4. **Audit Log Growth**
   - Ensure PaymentAuditLog is being written
   - Alert if no settlement logs for > 1 hour (during active hours)

### Dashboard Queries

```sql
-- Transactions needing reconciliation
SELECT COUNT(*) 
FROM PaymentTransaction 
WHERE purpose = 'wallet_topup' 
  AND status = 'paid' 
  AND settlementStatus != 'settled';

-- Settlement latency (how long from paid to settled)
SELECT 
  AVG(TIMESTAMPDIFF(SECOND, paidAt, settledAt)) as avg_seconds,
  MAX(TIMESTAMPDIFF(SECOND, paidAt, settledAt)) as max_seconds
FROM PaymentTransaction
WHERE purpose = 'wallet_topup'
  AND settlementStatus = 'settled'
  AND paidAt IS NOT NULL
  AND settledAt IS NOT NULL;

-- Audit log summary
SELECT 
  action,
  source,
  COUNT(*) as count
FROM PaymentAuditLog
WHERE createdAt > DATE_SUB(NOW(), INTERVAL 24 HOUR)
GROUP BY action, source;
```
